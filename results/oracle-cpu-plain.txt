def fibonacci(n: int, memo: Dict[int, int] = None) -> int:
    """
    Compute the nth Fibonacci number using memoization.

    Args:
        n: A non-negative integer representing the position in the Fibonacci sequence.
        memo: A dictionary used to cache previously computed Fibonacci numbers.

    Returns:
        The nth Fibonacci number.

    Raises:
        ValueError: If n is negative.

    Example:
        >>> fibonacci(10)
        55
        >>> fibonacci(0)
        0
        >>> fibonacci(1)
        1
    """
    if n < 0:
        raise ValueError("n must be a non-negative integer")

        memo = {}

    if n in memo:
        return memo[n]

    if n == 0:
        return 0
    if n == 1:
        return 1

    memo[n] = fibonacci(n - 1, memo) + fibonacci(n - 2, memo)
    return memo[n]
